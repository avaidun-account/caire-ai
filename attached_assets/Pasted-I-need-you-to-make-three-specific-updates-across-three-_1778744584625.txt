function buildConsensus(results) {
  const valid = results.filter(r => r.result);
  if (valid.length < 2) return;

  const urgencies = valid.map(r => r.result.urgency);
  const allAgree = urgencies.every(u => u === urgencies[0]);
  const maxDiff = Math.max(...urgencies) - Math.min(...urgencies);
  const allConsiderations = valid.flatMap(r => r.result.considerations || []);

  let dotClass, alignmentText;
  if (allAgree) {
    dotClass = 'dot-agree';
    alignmentText = `All 3 models agree on urgency: <strong>${valid[0].result.urgency_label}</strong>. This consistency increases confidence in the assessment.`;
  } else if (maxDiff === 1) {
    dotClass = 'dot-warn';
    alignmentText = `Models differ by one urgency tier. The most cautious recommendation is shown above. When in doubt, treat as the higher urgency.`;
  } else {
    dotClass = 'dot-disagree';
    alignmentText = `Models disagree significantly on urgency. This level of uncertainty is itself a reason to consult a healthcare professional promptly.`;
  }

  const countMap = {};
  allConsiderations.forEach(c => {
    const key = c.toLowerCase().slice(0, 40);
    if (!countMap[key]) countMap[key] = { text: c, count: 0 };
    countMap[key].count++;
  });
  const shared = Object.values(countMap)
    .filter(v => v.count >= 2)
    .map(v => v.text)
    .slice(0, 3);

  const allPoints = valid.map(r => ({
    model: r.model,
    points: r.result.considerations || []
  }));
  const uniquePoints = allPoints.map(m => ({
    model: m.model,
    unique: m.points.filter(p =>
      countMap[p.toLowerCase().slice(0, 40)]?.count === 1
    )
  })).filter(m => m.unique.length > 0);

  const container = document.getElementById('consensusRows');
  container.innerHTML = `
    <div class="consensus-block">
      <div class="consensus-block-title">Where models align</div>
      <div class="consensus-row">
        <div class="dot ${dotClass}"></div>
        <div class="consensus-text">${alignmentText}</div>
      </div>
      ${shared.length > 0 ? `
      <div class="consensus-row">
        <div class="dot dot-agree"></div>
        <div class="consensus-text"><strong>All models flagged:</strong> ${shared.join(' · ')}</div>
      </div>` : ''}
    </div>

    ${uniquePoints.length > 0 ? `
    <div class="consensus-block">
      <div class="consensus-block-title">Key differences</div>
      ${uniquePoints.map(m => `
      <div class="consensus-row">
        <div class="dot dot-warn"></div>
        <div class="consensus-text"><strong>${m.model} only flagged:</strong> ${m.unique.join('; ')}</div>
      </div>`).join('')}
    </div>` : ''}

    <div class="consensus-block">
      <div class="consensus-block-title">Actionable insights for caregivers</div>

        <div class="consensus-text"><strong>Watch for immediately:</strong> ${getEscalationTriggers(valid)}</div>

function getEscalationTriggers(valid) {
  const urgencyLevel = Math.min(...valid.map(r => r.result.urgency));
  if (urgencyLevel === 1) return 'Go to the ER immediately — do not wait.';
  const allText = valid.flatMap(r => r.result.considerations || []).join(' ').toLowerCase();
  if (allText.includes('er') || allText.includes('emergency') || allText.includes('911')) {
    return 'Any rapid worsening — go to the ER immediately if that occurs.';
  }
  return 'Any significant worsening, new symptoms, or gut feeling something is wrong — seek care sooner.';
}
